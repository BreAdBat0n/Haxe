Работа с DOM

При компиляции в JavaScript доступен модуль js.Browser, через который можно обращаться к DOM так же, как в обычном JS.

Получить элемент

import js.Browser;

var el = Browser.document.getElementById("app");

Создать элемент

import js.Browser;

var div = Browser.document.createElement("div");
div.textContent = "Привет!";
Browser.document.body.appendChild(div);

Изменить стиль

var el = Browser.document.getElementById("app");
el.style.color = "orange";
el.style.fontSize = "18px";

Добавить класс

el.classList.add("active");
el.classList.remove("active");
el.classList.toggle("active");

Обработчик события

import js.Browser;

var btn = Browser.document.getElementById("myBtn");
btn.addEventListener("click", function(e) {
  trace("клик!");
});

Ввод из поля

import js.html.InputElement;

var input = cast Browser.document.getElementById("name");
var value = input.value;

Таймеры

import js.Browser;

Browser.window.setTimeout(function() {
  trace("прошло 2 секунды");
}, 2000);

var id = Browser.window.setInterval(function() {
  trace("каждую секунду");
}, 1000);
Browser.window.clearInterval(id);

Работа с localStorage

import js.Browser;

Browser.window.localStorage.setItem("key", "value");
var v = Browser.window.localStorage.getItem("key");
Browser.window.localStorage.removeItem("key");

Типизированные элементы

Для типизации используйте js.html.*:

import js.html.DivElement;
import js.html.InputElement;

var div:DivElement = Browser.document.createElement("div");
var input:InputElement = cast Browser.document.getElementById("name");

Важно

Весь код с js.Browser работает только при компиляции в JavaScript. При сборке в C++ или Java этот код не скомпилируется — используйте условную компиляцию:

#if js
  import js.Browser;
  Browser.document.title = "Haxe";
#end
