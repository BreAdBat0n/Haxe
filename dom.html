Работа с DOM

При компиляции в JavaScript доступен модуль js.Browser, через который можно обращаться к DOM так же, как в обычном JS.

Получить элемент

import js.Browser;

var el = Browser.document.getElementById("app");
var byClass = Browser.document.querySelector(".card");
var all = Browser.document.querySelectorAll("div");

Создать элемент

import js.Browser;

var div = Browser.document.createElement("div");
div.textContent = "Привет!";
Browser.document.body.appendChild(div);

Вставить HTML

el.innerHTML = "<b>жирный</b>";
el.textContent = "просто текст";

innerHTML вставляет HTML, textContent — только текст (безопаснее).

Изменить стиль

var el = Browser.document.getElementById("app");
el.style.color = "orange";
el.style.fontSize = "18px";
el.style.display = "none";

Добавить класс

el.classList.add("active");
el.classList.remove("active");
el.classList.toggle("active");
el.classList.contains("active");

Атрибуты

el.setAttribute("data-id", "5");
var id = el.getAttribute("data-id");
el.removeAttribute("data-id");

Обработчик события

import js.Browser;

var btn = Browser.document.getElementById("myBtn");
btn.addEventListener("click", function(e) {
  trace("клик!");
});

Удалить обработчик

function onClick(e) { trace("клик"); }
btn.addEventListener("click", onClick);
btn.removeEventListener("click", onClick);

События клавиатуры

import js.html.KeyboardEvent;

Browser.document.addEventListener("keydown", function(e:KeyboardEvent) {
  if (e.key == "Escape") trace("Esc нажат");
});

Ввод из поля

import js.html.InputElement;

var input = cast Browser.document.getElementById("name");
var value = input.value;
input.value = "новое значение";

Формы

import js.html.FormElement;

var form:FormElement = cast Browser.document.getElementById("myForm");
form.addEventListener("submit", function(e) {
  e.preventDefault();
  trace("отправка");
});

Таймеры

import js.Browser;

Browser.window.setTimeout(function() {
  trace("прошло 2 секунды");
}, 2000);

var id = Browser.window.setInterval(function() {
  trace("каждую секунду");
}, 1000);
Browser.window.clearInterval(id);

Работа с localStorage

import js.Browser;

Browser.window.localStorage.setItem("key", "value");
var v = Browser.window.localStorage.getItem("key");
Browser.window.localStorage.removeItem("key");
Browser.window.localStorage.clear();

sessionStorage

Browser.window.sessionStorage.setItem("key", "value");

Fetch (HTTP-запросы)

import js.Browser;
import js.html.Response;

Browser.window.fetch("https://api.example.com/data")
  .then(function(res:Response) return res.json())
  .then(function(data) trace(data))
  .catchError(function(err) trace("ошибка: " + err));

XMLHttpRequest

import js.html.XMLHttpRequest;

var xhr = new XMLHttpRequest();
xhr.open("GET", "/api/data");
xhr.onload = function() {
  trace(xhr.responseText);
};
xhr.send();

Типизированные элементы

Для типизации используйте js.html.*:

import js.html.DivElement;
import js.html.InputElement;
import js.html.ButtonElement;

var div:DivElement = Browser.document.createElement("div");
var input:InputElement = cast Browser.document.getElementById("name");
var btn:ButtonElement = cast Browser.document.getElementById("btn");

Работа с URL

import js.html.Location;

var loc = Browser.window.location;
trace(loc.href);
trace(loc.pathname);
trace(loc.search);
loc.href = "https://example.com";

История браузера

Browser.window.history.pushState({ page: 1 }, "", "/page1");
Browser.window.history.back();

Canvas

import js.html.CanvasElement;
import js.html.CanvasRenderingContext2D;

var canvas:CanvasElement = cast Browser.document.getElementById("canvas");
var ctx:CanvasRenderingContext2D = canvas.getContext2d();
ctx.fillStyle = "orange";
ctx.fillRect(0, 0, 100, 100);

Важно: только для JS

Весь код с js.Browser работает только при компиляции в JavaScript. При сборке в C++ или Java этот код не скомпилируется — используйте условную компиляцию:

#if js
  import js.Browser;
  Browser.document.title = "Haxe";
#end
